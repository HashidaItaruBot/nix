创建 `nixos/openhands/podman-compose.yml` 文件：

```yaml
services:
  openhands:
    image: ghcr.io/openhands/openhands:latest
    container_name: openhands-app
    ports:
      - "3000:3000"
    volumes:
      - /var/run/podman/podman.sock:/var/run/docker.sock
      - /home/zhaochanglong/.openhands/data:/.openhands
      - /home/zhaochanglong/.openhands/workspace:/opt/workspace_base
    environment:
      - LLM_API_KEY=${LLM_API_KEY}
      - LLM_MODEL=claude-sonnet-4-5-20250929
      - WORKSPACE_MOUNT_PATH=/opt/workspace_base
    restart: unless-stopped
```

注意：
- 使用 `ghcr.io/openhands/openhands:latest` 镜像
- Podman socket 挂载到 `/var/run/docker.sock`（OpenHands 期望的路径）
- 环境变量使用 shell 变量替换
