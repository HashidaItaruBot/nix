```yaml
# nixos/openhands/docker-compose.yml
services:
  openhands:
    image: ghcr.io/all-hands-ai/openhands:main
    container_name: openhands-app
    ports:
      - "3000:3000"
    volumes:
      - /var/run/docker.sock:/var/run/docker.sock
      - /var/lib/openhands/data:/.openhands
      - /opt/workspace:/opt/workspace
    environment:
      - LLM_API_KEY=${LLM_API_KEY}
      - LLM_MODEL=${LLM_MODEL:-claude-sonnet-4-5-20250929}
      - WORKSPACE_MOUNT_PATH=/opt/workspace
    extra_hosts:
      - "host.docker.internal:host-gateway"
    restart: unless-stopped
```

- Docker socket 挂载用于沙箱容器创建

创建 `nixos/openhands/service.nix` 定义 systemd 服务：

```nix
# nixos/openhands/service.nix
{ config, pkgs, lib, ... }:
{
  services.openhands = {
    enable = lib.mkOption {
      type = lib.types.bool;
      default = false;
      description = "Enable OpenHands AI programming assistant";
    };
    workDir = lib.mkOption {
      type = lib.types.path;
      default = "/var/lib/openhands";
      description = "Working directory for OpenHands";
    };
    dataDir = lib.mkOption {
      type = lib.types.path;
      default = "/var/lib/openhands/data";
      description = "Data directory for persistent storage";
    };
    workspaceDir = lib.mkOption {
      type = lib.types.path;
      default = "/opt/workspace";
      description = "Workspace directory for code projects";
    };
  };

  systemd.services.openhands = {
    description = "OpenHands AI Programming Assistant";
    wantedBy = [ "multi-user.target" ];
    after = [ "docker.service" "network-online.target" ];
    requires = [ "docker.service" ];

    serviceConfig = {
      Type = "oneshot";
      RemainAfterExit = true;
      WorkingDirectory = "/var/lib/openhands";
      ExecStart = "${pkgs.docker-compose}/bin/docker-compose -f /var/lib/openhands/docker-compose.yml up -d";
      ExecStop = "${pkgs.docker-compose}/bin/docker-compose -f /var/lib/openhands/docker-compose.yml down";
      Restart = "on-failure";
      RestartSec = "10s";
    };

    environment = {
      DOCKER_HOST = "unix:///var/run/docker.sock";
    };
  };

  # 创建持久化目录
  systemd.tmpfiles.rules = [
    "d /var/lib/openhands 0755 root root -"
    "d /var/lib/openhands/data 0755 root root -"
    "d /opt/workspace 0755 root root -"
    "L+ /var/lib/openhands/docker-compose.yml - - - - /etc/nixos/openhands/docker-compose.yml"
  ];

  # 环境变量文件（由 sops 管理，Plan 03 配置）
  environment.etc."openhands.env".text = ''
    # Placeholder - actual values from sops in next plan
    LLM_API_KEY=${LLM_API_KEY:-your_api_key_here}
    LLM_MODEL=${LLM_MODEL:-claude-sonnet-4-5-20250929}
  '';
}
```
