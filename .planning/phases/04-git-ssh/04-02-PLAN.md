    ```bash
    #!/usr/bin/env bash
    set -e

    echo "=== Podman Machine Initialization ==="

    # 检查 machine 是否已存在
    if podman machine ls 2>/dev/null | grep -q "podman.*Running"; then
      echo "Machine already running, nothing to do"
      exit 0
    fi

    # 如果 machine 存在但未运行，启动它
    if podman machine ls 2>/dev/null | grep -q "podman"; then
      echo "Starting existing machine..."
      podman machine start
      exit 0
    fi

    # 首次初始化，带资源配置
    echo "Initializing Podman machine with 8GB RAM, 4 CPUs..."
    podman machine init --cpus 4 --memory 8192 --disk-size 100

    echo "Starting machine..."
    podman machine start

    echo "=== Initialization complete ==="
    ```

    创建后添加执行权限：chmod +x scripts/init-podman-machine.sh

    在现有配置中添加：
    ```nix
    launchd.user.agents.podman-start = {
      command = "${./scripts/init-podman-machine.sh}";
      serviceConfig = {
        KeepAlive = false;  # 只启动一次，不持续重启
        RunAtLoad = true;   # 登录时运行
        StandardOutPath = "/tmp/podman-start.log";
        StandardErrorPath = "/tmp/podman-start.err";
      };
    };
    ```

    注意：
    - 使用 ${./scripts/init-podman-machine.sh} 引用脚本（Nix 会计算绝对路径）
    - KeepAlive = false 避免重复启动
    - RunAtLoad = true 实现登录自启
    - 日志路径配置便于调试
