```nix
# nixos/microvm/openclaw.nix
# Guest 配置：定义 openclaw MicroVM 的 NixOS 配置
{ config, lib, pkgs, inputs, ... }: {
  # 导入 microvm.nix 的 microvm 模块
  # 提供 microvm.* 配置选项
  imports = [
    inputs.microvm.nixosModules.microvm
  ];

  # VM 主机名
  networking.hostName = "openclaw";

  # 核心 MicroVM 配置
  microvm = {
    # MIC-03: 使用 cloud-hypervisor 作为 hypervisor
    hypervisor = "cloud-hypervisor";

    # MIC-04: 分配 2GB 内存
    mem = 2048;

    # 配置 2 vCPU
    vcpu = 2;

    # 控制 socket 路径，用于优雅关闭 VM
    socket = "/var/lib/microvms/openclaw/control.sock";

    # virtiofs 只读共享宿主机的 /nix/store
    # 避免在 guest 镜像中重复打包 Nix 包
    shares = [{
      proto = "virtiofs";  # cloud-hypervisor 仅支持 virtiofs
      tag = "ro-store";
      source = "/nix/store";
      mountPoint = "/nix/.ro-store";
    }];

    # writable overlay 支持运行时写入（如 Playwright 浏览器下载）
    # 重启后 overlay 数据保留，但 Nix DB 可能忘记构建的包
    writableStoreOverlay = "/nix/.rw-store";
    volumes = [{
      image = "rw-store.img";
      mountPoint = config.microvm.writableStoreOverlay;
      size = 2048;  # 2GB
    }];

    # TAP 网络接口（NAT 模式，仅宿主机可访问）
    interfaces = [{
      type = "tap";
      id = "microvm-openclaw";  # TAP 接口名称
      mac = "02:00:00:00:00:01";  # 本地管理的 MAC 地址
    }];
  };

  # 安装必要的工具包
  # git: 代码版本控制
  # curl, wget: 网络工具
  # vim: 文本编辑
  # nodejs_22: Node.js 运行时（openclaw 必需）
  # python3, gcc, make: 编译原生模块
  # jq: JSON 处理
  # htop: 系统监控
  # xorg.xvfb: 无头 X11 显示（Playwright headed 模式）
  environment.systemPackages = with pkgs; [
    git
    curl
    wget
    vim
    nodejs_22
    python3
    gcc
    make
    jq
    htop
    xorg.xvfb
  ];

  # 配置 Xvfb  systemd 服务
  # 提供 :0 虚拟显示，用于 Playwright headed 模式测试
  systemd.services.xvfb = {
    description = "Xvfb Virtual Display";
    wantedBy = [ "multi-user.target" ];
    serviceConfig = {
      ExecStart = "${pkgs.xorg.xvfb}/bin/Xvfb :0 -screen 0 1280x1024x24";
      Restart = "on-failure";
    };
    environment = {
      DISPLAY = ":0";
    };
  };

  # 全局 DISPLAY 环境变量
  environment.variables = {
    DISPLAY = ":0";
  };

  # NixOS 状态版本
  system.stateVersion = "25.11";
}
```

关键点：
- 必须导入 `inputs.microvm.nixosModules.microvm`
- 必须配置 `hypervisor = "cloud-hypervisor"` (MIC-03)
- 必须配置 `mem = 2048` (MIC-04)
- 必须配置 `vcpu = 2`
- 必须配置 virtiofs shares 只读共享 /nix/store
- 必须配置 writableStoreOverlay 和 volumes
- 必须配置 TAP 网络接口
- 必须安装 nodejs_22 和其他工具包
- 必须配置 Xvfb systemd 服务
